153 uint8_t getCalibErrorInfo() const final
154 {
155 Zumo2040::LinesensorsInfo info;
156 info = m_lineSensors.getInfo();
157
158 if(info.m_isCalibrated == false)
159 {
160 return CALIB_ERROR_NOT_CALIBRATED;
161 }
162
163 for (uint8_t index = 0; index < MAX_SENSORS; ++index)
164 {
165 if(info.m_sensorError[index] == Zumo2040::LINESENSOR_CALIBRATION_FAILED)
166 {
167 return index;
168 }
169 }
170
171 return CALIB_ERROR_OK;
172 }
